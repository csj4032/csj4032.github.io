두 정수 A와 B를 입력받은 다음, A에서 B를 뺀 결과를 출력

```java
import java.util.Scanner;

/**
 * 제목 : A-B
 * 링크 : https://www.acmicpc.net/problem/1001
 * 단계 : 입/출력 받아보기
 * 분류 : 사칙연산, 수학
 */
public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int a = sc.nextInt();
        int b = sc.nextInt();
        System.out.println(a-b);
    }
}
```
